    const button = document.querySelector('[data-theme-toggle]');
    const label = document.querySelector('[data-theme-label]');
    const meta = document.querySelector('meta[name="theme-color"]');
    const apply = (theme) => {
      root.dataset.theme = theme;
      button?.setAttribute('aria-pressed', String(theme === 'carbon'));
      if (label) label.textContent = theme === 'carbon' ? 'light' : 'dark';
      if (meta) meta.content = theme === 'carbon' ? '#111216' : '#f6f5f1';
      try { localStorage.setItem('specimen-theme', theme); } catch {}
      window.dispatchEvent(new CustomEvent('specimen:theme', { detail: theme }));
    };
    apply(root.dataset.theme === 'carbon' ? 'carbon' : 'paper');
    button?.addEventListener('click', () => apply(root.dataset.theme === 'carbon' ? 'paper' : 'carbon'));
  }

  function setupClock() {
    let hour12 = false;
    const clock = document.querySelector('[data-clock]');
    const date = document.querySelector('[data-date]');
    const greeting = document.querySelector('[data-greeting]');
    const update = () => {
      const now = new Date();
      if (clock) clock.textContent = now.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', second: '2-digit', hour12 });
      if (date) date.textContent = now.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: '2-digit', year: 'numeric' }).toUpperCase();
      if (greeting) {
        const h = now.getHours();
        greeting.textContent = h < 5 ? 'up late?' : h < 12 ? 'good morning' : h < 18 ? 'good afternoon' : 'good evening';
      }
    };
    document.querySelector('[data-toggle-clock]')?.addEventListener('click', () => { hour12 = !hour12; update(); showToast(hour12 ? '12-hour clock.' : '24-hour clock.'); });
    document.querySelector('[data-copy-date]')?.addEventListener('click', () => copyText(new Date().toISOString().slice(0, 10), 'Date copied.'));
    greeting?.addEventListener('click', () => showToast('Welcome to the specimen.'));
    update();
    setInterval(update, 1000);
  }

  function setupReveal() {
    const items = document.querySelectorAll('[data-reveal]');
    if (reduceMotion.matches || !('IntersectionObserver' in window)) {
      items.forEach((item) => item.classList.add('is-visible'));
      return;
    }
    const observer = new IntersectionObserver((entries) => {
      entries.forEach((entry) => {
        if (!entry.isIntersecting) return;
        entry.target.classList.add('is-visible');
        observer.unobserve(entry.target);
      });
    }, { threshold: .12 });
    items.forEach((item) => observer.observe(item));
  }

  function setupStartup() {
    const startup = document.querySelector('[data-startup]');
    if (!startup || reduceMotion.matches) { startup?.remove(); return; }
    window.setTimeout(() => startup.classList.add('is-done'), 1250);
    window.setTimeout(() => startup.remove(), 1900);
  }

  function setupScrollRule() {
    const rule = document.querySelector('[data-scroll-rule]');
    let ticking = false;
    const update = () => {
      const max = document.documentElement.scrollHeight - innerHeight;
      const progress = max > 0 ? scrollY / max : 0;
      if (rule) rule.style.transform = `scaleX(${Math.min(1, Math.max(0, progress))})`;
      ticking = false;
    };
    addEventListener('scroll', () => { if (!ticking) { ticking = true; requestAnimationFrame(update); } }, { passive: true });
    update();
  }

  function setupCrosshair() {
    const crosshair = document.querySelector('[data-crosshair]');
    if (!crosshair || matchMedia('(hover: none)').matches || reduceMotion.matches) return;
    addEventListener('pointermove', (event) => {
      crosshair.style.left = `${event.clientX}px`;
      crosshair.style.top = `${event.clientY}px`;
      crosshair.classList.add('is-on');
    }, { passive: true });
    addEventListener('pointerleave', () => crosshair.classList.remove('is-on'));
  }

  function setupAmbientGrid() {
    const canvas = document.querySelector('[data-ambient-grid]');
    if (!canvas?.getContext) return;
    const ctx = canvas.getContext('2d');
    let width = 0, height = 0, dpr = 1, pointerX = -1000, pointerY = -1000, raf = 0;
    const resize = () => {
      dpr = Math.min(devicePixelRatio || 1, 2);
      width = innerWidth; height = innerHeight;
      canvas.width = Math.round(width * dpr); canvas.height = Math.round(height * dpr);
      canvas.style.width = `${width}px`; canvas.style.height = `${height}px`;
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    };
    const draw = (time = 0) => {
      ctx.clearRect(0, 0, width, height);
      const dark = root.dataset.theme === 'carbon';
      const spacing = width < 700 ? 34 : 42;
      for (let x = spacing / 2; x < width; x += spacing) {
        for (let y = spacing / 2; y < height; y += spacing) {
          const distance = Math.hypot(x - pointerX, y - pointerY);
          const near = Math.max(0, 1 - distance / 190);
          const pulse = reduceMotion.matches ? 0 : (Math.sin((x + y) * .017 + time * .0008) + 1) * .5;
          const alpha = .06 + near * .22 + pulse * .018;
          ctx.beginPath();
          ctx.arc(x, y, 1 + near * 1.6, 0, Math.PI * 2);
          ctx.fillStyle = dark ? `rgba(236,238,242,${alpha})` : `rgba(17,19,26,${alpha})`;
          ctx.fill();
        }
      }
      if (!reduceMotion.matches) raf = requestAnimationFrame(draw);
    };
    addEventListener('resize', resize, { passive: true });
    addEventListener('pointermove', (event) => { pointerX = event.clientX; pointerY = event.clientY; }, { passive: true });
    addEventListener('pointerleave', () => { pointerX = pointerY = -1000; });
    addEventListener('specimen:theme', () => draw(performance.now()));
    resize();
    draw();
  }
